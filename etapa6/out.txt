int a = 0;
int b = 1;
int next = 0;
int i = 0;
int count = 0;
int main () {
    print "Enter the number of Fibonacci numbers to generate: ";
    read count;
    count = print_fibonacci(count);
    return 0;
}
int print_fibonacci (int n) {
    while (i < n) do {
        print a "\n";
        next = a + b;
        a = b;
        b = next;
        i = i + 1;
    }
    print "\n";
}
